# ==================================================
# 날씨로 태양광 발전량을 예측하는 인공지능
# ==================================================
# Colab 첫 칸에서 한 번만 실행하세요:
# !pip install -q koreanize-matplotlib

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

try:
    import koreanize_matplotlib
except ImportError:
    pass

from sklearn.preprocessing import StandardScaler
from sklearn.metrics import mean_absolute_error, r2_score

import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Input
from tensorflow.keras.callbacks import EarlyStopping

# 실행할 때마다 같은 결과가 나오도록 난수 고정
tf.keras.utils.set_random_seed(42)


# ==================================================
# 파일 업로드 (FileNotFoundError 해결을 위해 추가)
# ==================================================
from google.colab import files
# `files.upload()`는 파일이 이미 존재하므로 제거되었습니다.
# uploaded = files.upload()

# ==================================================
# 1단계. 엑셀 파일 읽기
# ==================================================

# NOTE: The filename below should match the name of the file you uploaded.
# If you uploaded a file with a different name, please update the path accordingly.
data = pd.read_excel(
    "/content/solar.xlsx",
    sheet_name="날짜일시_옆으로통합"
)
# 날짜를 컴퓨터가 알아보는 형식으로 바꾸고,
# 날짜 순서대로 정렬
data["일시"] = pd.to_datetime(data["일시"])
data = data.sort_values("일시").reset_index(drop=True)

# 몇 시인지 따로 뽑기
data["시각"] = data["일시"].dt.hour


# ==================================================
# 2단계. 데이터 정리하기
# ==================================================

# 강수량이 빈칸이면 비가 오지 않은 것으로 보고 0으로 채운다.
data["기상_강수량(mm)"] = data["기상_강수량(mm)"].fillna(0)

# 엑셀의 '전체합'에는 잘못 들어간 숫자가 섞여 있고,
# 인버터1은 5월 중순부터 고장 나서 계속 0이므로
# 정상적으로 작동한 인버터 2, 3, 4의 발전량을 더해서 사용한다.
data["발전량"] = (
    data["인버터2(kWh)"]
    + data["인버터3(kWh)"]
    + data["인버터4(kWh)"]
)


# ==================================================
# 3단계. 입력과 정답 정하기
# ==================================================

# AI에게 알려 줄 정보 = 날씨 + 시각
입력 = [
    "기상_일사(MJ/m2)",       # 햇빛의 양
    "기상_기온(°C)",          # 온도
    "기상_습도(%)",           # 습도
    "기상_풍속(m/s)",         # 바람의 세기
    "기상_전운량(10분위)",    # 구름의 양 (0 = 맑음, 10 = 흐림)
    "기상_현지기압(hPa)",     # 기압
    "기상_강수량(mm)",        # 비가 내린 양
    "시각",                   # 시간
]

# AI가 예측해야 할 정답
정답 = "발전량"

print("전체 데이터 개수:", len(data))


# ==================================================
# 4단계. 학습 데이터와 평가 데이터 나누기
# ==================================================

# 과거 데이터 80%를 AI 학습에 사용하고,
# 이후 날짜의 20%는 학습에 사용하지 않고
# 최종 평가에 사용한다.
#
# 시간 순서를 섞지 않는 이유:
# 실제 상황에서는 미래의 날씨와 발전량을 미리 알 수 없기 때문이다.
# 따라서 과거 데이터를 학습하고
# 이후의 데이터를 예측하는 방식으로 평가한다.

나누는곳 = int(len(data) * 0.8)

# 하루 중간에서 데이터가 잘리지 않도록
# 다음 날 아침 7시부터 평가가 시작되게 맞춘다.
while data["시각"][나누는곳] != 7:
    나누는곳 = 나누는곳 + 1

학습 = data[:나누는곳]
평가 = data[나누는곳:]

print("학습 데이터:", len(학습), "개 / 평가 데이터:", len(평가), "개")
print(
    "평가 기간:",
    평가["일시"].min().date(),
    "~",
    평가["일시"].max().date()
)


# ==================================================
# 5단계. 숫자 크기 맞추기 (표준화)
# ==================================================

# 기압은 1000 정도이고,
# 강수량은 0~10 정도처럼 변수마다 숫자의 크기가 다르다.
# AI가 숫자의 크기 때문에 특정 변수에 치우치지 않도록
# 모든 입력값의 크기를 비슷하게 맞춘다.
#
# 기준은 학습 데이터로만 만든다.

scaler = StandardScaler()

X_학습 = scaler.fit_transform(학습[입력])
X_평가 = scaler.transform(평가[입력])

y_학습 = 학습[정답]
y_평가 = 평가[정답]


# ==================================================
# 6단계. 인공지능(인공 신경망) 만들기
# ==================================================

# 입력 8개
#     ↓
# 뉴런 32개
#     ↓
# 뉴런 16개
#     ↓
# 발전량 1개

model = Sequential([
    Input(shape=(len(입력),)),
    Dense(32, activation="relu"),
    Dense(16, activation="relu"),
    Dense(1),
])

# adam: 오차를 줄이도록 가중치를 조정하는 방법
# mse : 예측값과 실제값의 차이를 계산하는 방법
model.compile(
    optimizer="adam",
    loss="mse"
)


# ==================================================
# 7단계. AI 학습시키기
# ==================================================

# 학습 데이터의 20%는 학습에 직접 사용하지 않고
# '검증 데이터'로 사용한다.
#
# 학습하는 동안 검증 데이터에서도 오차가 줄어드는지 확인해서
# AI가 학습 데이터를 지나치게 외우고 있는지 확인한다.
#
# 검증 데이터의 오차가 15번 연속 좋아지지 않으면
# 학습을 멈춘다.
# 이를 통해 불필요하게 오래 학습하는 것을 줄인다.

조기종료 = EarlyStopping(
    monitor="val_loss",
    patience=15,
    restore_best_weights=True
)

history = model.fit(
    X_학습,
    y_학습,
    epochs=300,               # 최대 300번 반복해서 학습
    batch_size=32,            # 한 번에 32개씩 학습
    validation_split=0.2,     # 학습 데이터의 20%를 검증 데이터로 사용
    callbacks=[조기종료],
    verbose=0,
)

print(
    "실제로 학습한 횟수:",
    len(history.history["loss"]),
    "번"
)


# ==================================================
# 8단계. 학습하지 않은 데이터로 발전량 예측
# ==================================================

# AI가 학습에 사용하지 않은 이후 날짜의 데이터를 이용해서
# 실제 발전량을 예측한다.

예측 = model.predict(
    X_평가,
    verbose=0
).flatten()

# 발전량은 음수가 될 수 없으므로
# 음수로 예측된 값은 0으로 바꾼다.
예측 = np.maximum(예측, 0)


# 예측 결과를 표로 만든다.
결과 = pd.DataFrame({
    "일시": 평가["일시"].values,
    "실제 발전량": y_평가.values,
    "AI 예측": 예측.round(1),
})

# 예측값과 실제값의 차이
결과["오차"] = (
    결과["AI 예측"] - 결과["실제 발전량"]
).round(1)

print()
print(결과.head(12))

# 결과를 CSV 파일로 저장
결과.to_csv(
    "AI_발전량_예측결과.csv",
    index=False,
    encoding="utf-8-sig"
)


# ==================================================
# 9단계. AI 성능 평가하기
# ==================================================

# 평균 절대 오차(MAE):
# AI의 예측이 실제 발전량과 평균적으로
# 얼마나 차이가 나는지를 나타낸다.
# 값이 작을수록 예측이 정확하다.
#
# 결정계수(R²):
# 실제 발전량의 변화를 AI가 얼마나 잘 설명하는지를 나타낸다.
# 1에 가까울수록 실제 값과 예측 값이 잘 맞는다.

평균오차 = mean_absolute_error(
    y_평가,
    예측
)

결정계수 = r2_score(
    y_평가,
    예측
)

print()
print("====== AI 성능 평가 ======")
print(f"평균 절대 오차 : {평균오차:.1f} kWh")
print(f"결정계수(R²)   : {결정계수:.2f}")
print(
    f"(참고) 평가 기간 평균 발전량 : "
    f"{y_평가.mean():.1f} kWh"
)


# ==================================================
# 10단계. 그래프 ① 실제 발전량과 AI 예측 비교

# 평가 데이터 중 첫 5일을 보여준다.

보기 = 결과.head(60)

plt.figure(figsize=(12, 5))

# 시각을 제외하고 날씨 요소만 분석한다。
날씨 = 입력[:-1]

# 각 날씨 요소와 발전량의 상관계수를 계산한다.

# ==================================================
# 13단계. 새로운 날씨 조건으로 발전량 예측하기
# ==================================================

# 사용자가 직접 날씨 조건을 입력하면
# AI가 예상 발전량을 계산해 주는 함수

def 발전량_예측(
    일사량,
    기온,
    습도,
    풍속,
    구름양,
    기압,
    강수량,
    시각
):

    # 입력한 값을 표 한 줄로 만든다.
    # 입력 순서는 학습할 때 사용한 '입력' 목록과 같아야 한다.

    조건 = pd.DataFrame(
        [[
            일사량,
            기온,
            습도,
            풍속,
            구름양,
            기압,
            강수량,
            시각
        ]],
        columns=입력
    )

    # AI가 학습한 범위를 벗어난 값인지 확인한다。
    for 이름 in 입력:

        값 = 조건[이름][0]

        최소 = data[이름].min()
        최대 = data[이름].max()

        if 값 < 최소 or 값 > 최대:

            print(
                f"  ⚠ {이름} = {값} "
                f"은(는) 학습한 범위 "
                f"({최소} ~ {최대}) 밖이라 "
                f"예측이 부정확할 수 있습니다."
            )

    # 학습할 때 사용했던 것과 같은 기준으로
    # 숫자의 크기를 맞춘다。

    조건_변환 = scaler.transform(조건)

    # AI에게 발전량을 예측하게 한다.
    값 = model.predict(
        조건_변환,
        verbose=0
    )[0][0]

    # 발전량은 음수가 될 수 없으므로
    # 음수가 나오면 0으로 바꾼다.

    return max(
        float(값),
        0
    )

# ==================================================
# 15단계. 직접 날씨 조건을 입력해서 예측하기
# ==================================================

# 실행하면 각 날씨 조건을 직접 입력할 수 있다.
#
# 그만하려면 일사량 입력 칸에 q를 입력한다.

print()
print("====== 직접 입력해서 발전량 예측하기 ======")

print(
    "(참고 범위)"
)

print(
    "일사량 0~4, "
    "기온 3~38, "
    "습도 11~99, "
    "풍속 0~6, "
    "구름양 0~10, "
    "기압 986~1013, "
    "강수량 0~14, "
    "시각 7~18"
)


while True:

    일사량 = input(
        "\n일사량(MJ/m2) [그만하려면 q] : "
    )

    if 일사량.lower() == "q":
        break

    기온 = input("기온(°C)       : ")
    습도 = input("습도(%)        : ")
    풍속 = input("풍속(m/s)      : ")
    구름양 = input("구름양(0~10)   : ")
    기압 = input("기압(hPa)       : ")
    강수량 = input("강수량(mm)      : ")
    시각 = input("시각(7~18시)    : ")


    결과값 = 발전량_예측(
        float(일사량),
        float(기온),
        float(습도),
        float(풍속),
        float(구름양),
        float(기압),
        float(강수량),
        float(시각)
    )


    print(
        f"👉 AI 예측 발전량 : "
        f"{결과값:.1f} kWh"
    )
